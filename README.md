        /* Hàm tạo thẻ HTML cho từng sản phẩm chuẩn Flatsome:
           - Chỉ có duy nhất badge % giảm giá góc trên bên trái (.flatsome-sale-badge)
           - Không gắn thêm nhãn dán thừa
           - Tên sản phẩm hiển thị đầy đủ, không bị cắt chữ hay tràn ra ngoài */
        function createProductCardHTML(item) {
            const driveThumb = `https://drive.google.com/thumbnail?id=${item.driveId}&sz=w800`;
            return `
                <div class="flatsome-product-card group cursor-pointer" onclick="openProductModal(${item.id})">
                    <div class="relative w-full aspect-square bg-slate-100 overflow-hidden">
                        <span class="flatsome-sale-badge">${item.discount}</span>
                        <img src="${driveThumb}"
                             data-alt-src="https://lh3.googleusercontent.com/d/${item.driveId}=w800"
                             data-fallback="${item.fallback}"
                             onerror="if(this.src !== this.dataset.altSrc){this.src=this.dataset.altSrc;}else{this.onerror=null;this.src=this.dataset.fallback;}"
                             alt="${item.name}"
                             loading="lazy"
                             class="w-full h-full object-cover group-hover:scale-105 transition duration-300">
                    </div>
                    <div class="p-2.5 sm:p-3 flex flex-col flex-1 justify-between">
                        <div>
                            <div class="text-[10.5px] text-slate-400 font-medium mb-0.5">${item.subCat}</div>
                            <h3 class="flatsome-product-title group-hover:text-hvNavy transition">${item.name}</h3>
                        </div>
                        <div class="pt-1.5 border-t border-slate-100 mt-1">
                            <div class="flex flex-wrap items-baseline gap-1.5">
                                <del class="text-[11px] text-slate-400">${item.oldPrice}</del>
                                <span class="text-xs sm:text-sm font-extrabold text-hvRed">${item.price}</span>
                            </div>
                        </div>
                    </div>
                </div>
            `;
        }

        function renderAllSections(searchKeyword = '', subFilterGroup = null, subFilterVal = 'all') {
            const kw = searchKeyword.trim().toLowerCase();

            const renderGroup = (sectionKey, containerId) => {
                const container = document.getElementById(containerId);
                if (!container) return;

                const filtered = HIENVAN_PRODUCTS.filter(p => {
                    if (p.section !== sectionKey) return false;
                    if (kw && !p.name.toLowerCase().includes(kw) && !p.subCat.toLowerCase().includes(kw)) return false;
                    if (subFilterGroup === sectionKey && subFilterVal !== 'all' && p.subCat !== subFilterVal && p.catSlug !== subFilterVal) {
                        return false;
                    }
                    return true;
                });

                if (filtered.length === 0) {
                    container.innerHTML = `<div class="col-span-full py-6 text-center text-xs text-slate-500 bg-white rounded border border-slate-200">Không tìm thấy sản phẩm phù hợp. Hãy chọn "Tất cả" để xem đầy đủ.</div>`;
                } else {
                    container.innerHTML = filtered.map(createProductCardHTML).join('');
                }
            };

            renderGroup('tile', 'grid-tiles');
            renderGroup('sanitary', 'grid-sanitary');
            renderGroup('paint', 'grid-paint');
        }

        /* Mở Modal chi tiết sản phẩm */
        window.openProductModal = function(productId) {
            const item = HIENVAN_PRODUCTS.find(p => p.id === productId);
            if (!item) return;

            const modal = document.getElementById('productModal');
            document.getElementById('modalTitle').textContent = item.name;
            document.getElementById('modalBadge').textContent = item.discount;
            document.getElementById('modalCat').textContent = 'Phân loại: ' + item.subCat;
            document.getElementById('modalOldPrice').textContent = item.oldPrice;
            document.getElementById('modalPrice').textContent = item.price;
            document.getElementById('modalDriveLink').href = `https://drive.google.com/file/d/${item.driveId}/view?usp=sharing`;

            const imgEl = document.getElementById('modalImg');
            imgEl.src = `https://drive.google.com/thumbnail?id=${item.driveId}&sz=w1000`;
            imgEl.onerror = function() {
                this.onerror = null;
                this.src = item.fallback;
            };

            modal.classList.remove('hidden');
            modal.classList.add('flex');
        };

        document.addEventListener('DOMContentLoaded', function() {
            // Khởi tạo hiển thị toàn bộ 19 sản phẩm thực tế
            renderAllSections();

            // Đóng modal sản phẩm
            const productModal = document.getElementById('productModal');
            document.getElementById('closeModalBtn').addEventListener('click', () => {
                productModal.classList.add('hidden');
                productModal.classList.remove('flex');
            });
            productModal.addEventListener('click', (e) => {
                if (e.target === productModal) {
                    productModal.classList.add('hidden');
                    productModal.classList.remove('flex');
                }
            });

            // Xử lý lọc sub-category pills trên từng section
            document.querySelectorAll('.subcat-pill').forEach(pill => {
                pill.addEventListener('click', function() {
                    const group = this.dataset.filterGroup;
                    const sub = this.dataset.sub;
                    document.querySelectorAll(`.subcat-pill[data-filter-group="${group}"]`).forEach(el => el.classList.remove('active'));
                    this.classList.add('active');
                    renderAllSections('', group, sub);
                });
            });

            // Xử lý click vào các ô Danh mục chủ lực để cuộn mượt và lọc
            document.querySelectorAll('.cat-box-btn').forEach(btn => {
                btn.addEventListener('click', function() {
                    const cat = this.dataset.cat;
                    if (cat === 'all') {
                        renderAllSections();
                        document.getElementById('sec-gach-op-lat').scrollIntoView({ behavior: 'smooth' });
                    } else if (cat === 'thiet-bi-ve-sinh') {
                        renderAllSections();
                        document.getElementById('sec-thiet-bi-ve-sinh').scrollIntoView({ behavior: 'smooth' });
                    } else if (cat === 'son-nuoc' || cat === 'gach-trang-tri') {
                        renderAllSections();
                        document.getElementById('sec-son-nuoc').scrollIntoView({ behavior: 'smooth' });
                    } else {
                        renderAllSections();
                        document.getElementById('sec-gach-op-lat').scrollIntoView({ behavior: 'smooth' });
                    }
                });
            });

            // Tìm kiếm trên máy tính & điện thoại
            const handleSearch = (inputId) => {
                const val = document.getElementById(inputId).value;
                renderAllSections(val);
                document.getElementById('sec-gach-op-lat').scrollIntoView({ behavior: 'smooth' });
            };
            document.getElementById('desktopSearchBtn').addEventListener('click', () => handleSearch('desktopSearchInput'));
            document.getElementById('desktopSearchInput').addEventListener('keyup', (e) => {
                if (e.key === 'Enter') handleSearch('desktopSearchInput');
                else renderAllSections(e.target.value);
            });
            document.getElementById('mobileSearchBtn').addEventListener('click', () => handleSearch('mobileSearchInput'));
            document.getElementById('mobileSearchInput').addEventListener('keyup', (e) => {
                if (e.key === 'Enter') handleSearch('mobileSearchInput');
                else renderAllSections(e.target.value);
            });

            // Điều khiển Banner Slider 2 ảnh tự trượt
            let currentSlide = 0;
            const slides = [document.getElementById('slide-0'), document.getElementById('slide-1')];
            const dots = document.querySelectorAll('.banner-dot');

            function showSlide(index) {
                currentSlide = (index + slides.length) % slides.length;
                slides.forEach((sl, idx) => {
                    if (idx === currentSlide) sl.classList.add('active');
                    else sl.classList.remove('active');
                });
                dots.forEach((d, idx) => {
                    if (idx === currentSlide) {
                        d.className = 'banner-dot w-6 h-2 rounded-full bg-hvGold transition';
                    } else {
                        d.className = 'banner-dot w-2.5 h-2 rounded-full bg-white/60 transition';
                    }
                });
            }

            let bannerInterval = setInterval(() => showSlide(currentSlide + 1), 5000);
            const resetBannerTimer = () => {
                clearInterval(bannerInterval);
                bannerInterval = setInterval(() => showSlide(currentSlide + 1), 5000);
            };

            document.getElementById('prevBannerBtn').addEventListener('click', () => {
                showSlide(currentSlide - 1);
                resetBannerTimer();
            });
            document.getElementById('nextBannerBtn').addEventListener('click', () => {
                showSlide(currentSlide + 1);
                resetBannerTimer();
            });
            dots.forEach(dot => {
                dot.addEventListener('click', function() {
                    showSlide(parseInt(this.dataset.slide, 10));
                    resetBannerTimer();
                });
            });

            // Menu Drawer trên điện thoại
            const mobileMenuOverlay = document.getElementById('mobileMenuOverlay');
            document.getElementById('mobileMenuBtn').addEventListener('click', () => {
                mobileMenuOverlay.classList.remove('hidden');
            });
            document.getElementById('closeMobileMenuBtn').addEventListener('click', () => {
                mobileMenuOverlay.classList.add('hidden');
            });
            mobileMenuOverlay.addEventListener('click', (e) => {
                if (e.target === mobileMenuOverlay) mobileMenuOverlay.classList.add('hidden');
            });
            document.querySelectorAll('.mobile-cat-item').forEach(link => {
                link.addEventListener('click', () => {
                    mobileMenuOverlay.classList.add('hidden');
                });
            });

            // Xử lý gửi Form Yêu Cầu Báo Giá ở Trang Chủ
            document.getElementById('quoteForm').addEventListener('submit', function(e) {
                e.preventDefault();
                const name = document.getElementById('qName').value.trim();
                const phone = document.getElementById('qPhone').value.trim();
                const cat = document.getElementById('qCategory').value;
                const box = document.getElementById('quoteSuccessBox');

                box.innerHTML = `<i class="fa-solid fa-circle-check mr-1.5"></i>Cảm ơn <strong>${name}</strong> (${phone})! Yêu cầu báo giá hạng mục <strong>"${cat}"</strong> đã được ghi nhận. Bộ phận kinh doanh Nội Thất Hiền Vân sẽ liên hệ qua số <strong>0905.309.096</strong> ngay lập tức.`;
                box.classList.remove('hidden');
                this.reset();
            });
        });
